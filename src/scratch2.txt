    #tmp = zeros(length(u))
    #u = big.(v)
    for i = 1:length(ppairs)
        P(x) = pdf(d(ppairs[i][1],1),x)
        Q(x) = pdf(d(ppairs[i][2],1),x)
        Threads.@threads for j = 1:length(u)
            #@. tmp[j] = log(10,P(big(@view(u[j])))/Q(big(@view(u[j]))))*P(big(@view(u[j])))
            y[i,j] = @.  log(10,P(big(@view(u[j])))/Q(big(@view(u[j]))))*P(big(@view(u[j])))
        end
    end
end



m = 100;s = sort!(vcat(quantile(Exponential(10),(1:m)/(m+1)),-quantile(Exponential(10),(1:m)/(m+1))));n = length(s)
ppairs = SVector{Int(n*(n-1)/2)}([(s[i],s[j]) for i = 1:length(s) for j = (i+1):length(s)]);
 prototype_batch = (zeros(length(ppairs),1));

end
